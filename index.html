channel_revenue_attribution.sql — GA Sample Dataset (BigQuery Public) BIGQUERY 
-- Channel attribution: revenue and conversion rate by traffic source
-- Identifies highest-ROI acquisition channels for budget reallocation

SELECT
  trafficSource.source                          AS channel,
  trafficSource.medium                          AS medium,
  COUNT(DISTINCT fullVisitorId)                 AS unique_visitors,
  COUNT(DISTINCT visitId)                        AS sessions,
  SUM(totals.transactions)                       AS total_orders,
  ROUND(SUM(totals.transactionRevenue) / 1e6, 2)  AS revenue_usd,
  ROUND(
    SUM(totals.transactions) * 100.0 /
    NULLIF(COUNT(DISTINCT visitId), 0), 2
  )                                             AS conversion_rate_pct

FROM `bigquery-public-data.google_analytics_sample.ga_sessions_*`
WHERE _TABLE_SUFFIX BETWEEN '20170101' AND '20170731'
  AND totals.transactions IS NOT NULL
GROUP BY channel, medium
HAVING total_orders > 10
ORDER BY revenue_usd DESC
LIMIT 10;

supplier_performance_analysis.sql — Representative Logic BIGQUERY 
-- Supplier fulfillment performance: promised vs actual lead times
-- Identifies which suppliers are causing stockout risk through delivery delays

WITH supplier_metrics AS (
  SELECT
    supplier_name,
    COUNT(*)                                          AS total_orders,
    AVG(promised_lead_days)                            AS avg_promised_days,
    AVG(actual_lead_days)                              AS avg_actual_days,
    AVG(actual_lead_days - promised_lead_days)         AS avg_delay_days,
    COUNTIF(caused_stockout = TRUE)                   AS stockout_incidents
  FROM purchase_orders
  WHERE order_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 12 MONTH)
  GROUP BY supplier_name
)
SELECT *,
  CASE
    WHEN stockout_incidents > 2 OR avg_delay_days > 5  THEN 'HIGH RISK'
    WHEN avg_delay_days > 2                             THEN 'MONITOR'
    ELSE                                                     'RELIABLE'
  END AS risk_rating
FROM supplier_metrics
ORDER BY stockout_incidents DESC, avg_delay_days DESC;
